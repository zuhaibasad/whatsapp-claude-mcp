declare const sock: Record<string, () => Promise<void>>;

export async function unlink(): Promise<void> {
  await sock.logout();
}

// Right file, wrong function.
export async function disconnect(): Promise<void> {
  await sock.logout();
}

export async function viaElement(): Promise<void> {
  await sock['logout']();
}
